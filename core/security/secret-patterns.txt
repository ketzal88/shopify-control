# Secret patterns for core/hooks/scripts/secret-scan.sh
# One regex per line (grep -iE syntax). Lines starting with # are comments.
# Blank lines are ignored.
# Add project-specific patterns at the bottom.

# PEM private keys
BEGIN (RSA |EC |OPENSSH )?PRIVATE KEY

# AWS access key IDs
AKIA[0-9A-Z]{16}

# Google OAuth client secrets (service account JSON)
"client_secret"\s*:\s*"[A-Za-z0-9_\-]+

# Slack tokens (bot, app, user, legacy)
xox[baprs]-[0-9A-Za-z\-]+

# Anthropic API keys
sk-ant-[A-Za-z0-9\-_]+

# OpenAI API keys (also used by compatible providers)
sk-[A-Za-z0-9]{32,}

# Generic key / token / secret / password assignments with a quoted literal value.
# Charset includes / + = so base64 values match too.
(api[_-]?key|apikey|access[_-]?token|auth[_-]?token|client[_-]?secret|secret[_-]?key|password|passwd)["']?\s*[=:]\s*["'][A-Za-z0-9_./+=\-]{16,}

# Unquoted hex constants assigned to TOKEN/SECRET/KEY names (e.g. .env-style lines
# pasted into code). A real leak was a 32-hex token that a 24+ pattern elsewhere
# did not anchor; keep the bound loose (20+) and the name list explicit.
(token|secret|api[_-]?key)[A-Za-z0-9_]*\s*[=:]\s*[A-Fa-f0-9]{20,}

# Google API keys (Maps, Firebase web config, Gemini, ...)
AIza[0-9A-Za-z_\-]{35}

# Meta long-lived access tokens
EAAl[A-Za-z0-9]+

# GitHub personal access tokens (classic and fine-grained)
ghp_[A-Za-z0-9]{36}
github_pat_[A-Za-z0-9_]{82}

# Stripe secret keys
sk_live_[A-Za-z0-9]+
sk_test_[A-Za-z0-9]+
rk_live_[A-Za-z0-9]+

# Shopify Admin API / app tokens
shp(at|ss|ca|pa)_[A-Fa-f0-9]{32}

# Google API keys
AIza[0-9A-Za-z_-]{35}
